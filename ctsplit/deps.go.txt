package main

import (
	"fmt"
	"go/ast"
	"go/parser"
	"go/token"
	"os"
	"path/filepath"
	"sort"
	"strings"
)

func main() {
	dir := os.Args[1]
	fs := token.NewFileSet()
	files, _ := filepath.Glob(filepath.Join(dir, "*.go"))
	def := map[string]string{}
	parsed := map[string]*ast.File{}
	for _, f := range files {
		if strings.HasSuffix(f, "_test.go") {
			continue
		}
		af, err := parser.ParseFile(fs, f, nil, 0)
		if err != nil {
			panic(err)
		}
		b := filepath.Base(f)
		parsed[b] = af
		for _, d := range af.Decls {
			switch d := d.(type) {
			case *ast.FuncDecl:
				name := d.Name.Name
				if d.Recv != nil {
					name = "." + name
				}
				if _, ok := def[name]; !ok { def[name] = b }
			case *ast.GenDecl:
				for _, s := range d.Specs {
					switch s := s.(type) {
					case *ast.TypeSpec:
						def[s.Name.Name] = b
					case *ast.ValueSpec:
						for _, n := range s.Names {
							def[n.Name] = b
						}
					}
				}
			}
		}
	}
	for b, af := range parsed {
		deps := map[string]bool{}
		ast.Inspect(af, func(n ast.Node) bool {
			switch n := n.(type) {
			case *ast.Ident:
				if f, ok := def[n.Name]; ok && f != b {
					deps[f] = true
				}
			case *ast.SelectorExpr:
				if f, ok := def["."+n.Sel.Name]; ok && f != b {
					deps[f] = true
				}
			}
			return true
		})
		var ds []string
		for d := range deps {
			ds = append(ds, d)
		}
		sort.Strings(ds)
		fmt.Printf("%s -> %s\n", b, strings.Join(ds, " "))
	}
}
